```yaml
name: Autogenous Portfolio Sync (Code Commits)

on:
  push:
    branches:
      - main

jobs:
  sync-chronicle:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Codebase
        uses: actions/checkout@v4
        with:
          fetch-depth: 2 # Access previous state to check file diffs

      - name: Setup Python Runtime
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install Dependencies
        run: |
          pip install google-generativeai pymongo

      - name: Run Chronicler Compilation
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          MONGO_URI: ${{ secrets.MONGO_URI }}
        run: |
          python - <<EOF
          import os
          import sys
          import subprocess
          import time
          import google.generativeai as genai
          from pymongo import MongoClient

          commit_msg = ""
          diff_summary = ""

          # Pull Git metadata from the latest commit
          try:
              commit_msg = subprocess.check_output(["git", "log", "-1", "--pretty=%B"]).decode("utf-8").strip()
              diff_summary = subprocess.check_output(["git", "diff", "--stat", "HEAD~1", "HEAD"]).decode("utf-8").strip()
          except Exception as e:
              print(f"Warning: Could not extract full diff profile ({e}). Running on standard merge commit.")
              commit_msg = "Repository synchronized to main branch."

          # Support manual bypass tag
          if "[nosync]" in commit_msg:
              print("Bypass flag found. Sync aborted.")
              sys.exit(0)

          # Confirm environments
          api_key = os.getenv("GEMINI_API_KEY")
          mongo_uri = os.getenv("MONGO_URI")
          if not api_key or not mongo_uri:
              print("Error: Missing database credentials or Gemini API keys in repository secrets.")
              sys.exit(1)

          # API Initialization
          genai.configure(api_key=api_key)
          mongo_client = MongoClient(mongo_uri)
          db = mongo_client["portfolio"]

          system_prompt = (
              "You are the 'Chronicler Vessel' of Project Emergence. Your task is to look at a raw codebase commit message "
              "and the modified files, then translate this into a crisp, professional, job-ready resume accomplishment sentence.\n\n"
              "Rules:\n"
              "- Convert esoteric names into enterprise systems/operations terminology.\n"
              "- Ensure the sentence begins with an active engineering verb.\n"
              "- Focus on automation, orchestration scale (1,088 AI agents), and tech stack execution.\n"
              "- Output ONLY the final summary. Max 2 sentences."
          )

          model = genai.GenerativeModel(
              model_name='gemini-1.5-flash',
              system_instruction=system_prompt
          )
          
          prompt = f"Source: git | Commit Message: {commit_msg} | File changes: {diff_summary}"
          
          try:
              response = model.generate_content(prompt)
              summary = response.text.strip()

              # Commit clean entry into the MongoDB Stream collection
              db["chronicle_stream"].insert_one({
                  "timestamp": int(time.time()),
                  "source": "git",
                  "event_type": "push",
                  "raw_payload": {
                      "commit_message": commit_msg,
                      "file_stats": diff_summary
                  },
                  "executive_summary": summary
              })
              print(f"[+] Successfully posted translated commit stream: {summary}")
          except Exception as e:
              print(f"[X] API transmission bottleneck: {e}")
              sys.exit(1)
          EOF

```
